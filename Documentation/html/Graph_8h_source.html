370template <class T>
371bool Vertex<T>::removeEdge(T in) {
372 bool removedEdge = false;
373 auto it = adj.begin();
374 while (it != adj.end()) {
375 Edge<T> *edge = *it;
376 Vertex<T> *dest = edge->getDest();
377 if (dest->getInfo() == in) {
378 it = adj.erase(it);
379 deleteEdge(edge);
380 removedEdge = true; // allows for multiple edges to connect the same pair of vertices (multigraph)
381 }
382 else {
383 it++;
384 }
385 }
386 return removedEdge;
387}

391 auto it = adj.begin();
392 while (it != adj.end()) {
393 Edge<T> *edge = *it;
394 it = adj.erase(it);
395 deleteEdge(edge);
396 }
397}

596template <class T>
597bool Graph<T>::removeVertex(const T &in) {
598 for (auto it = vertexSet.begin(); it != vertexSet.end(); it++) {
599 if ((*it)->getInfo() == in) {
600 auto v = *it;
601 v->removeOutgoingEdges();
602 for (auto u : vertexSet) {
603 u->removeEdge(v->getInfo());
604 }
605 vertexSet.erase(it);
606 delete v;
607 return true;
608 }
609 }
610 return false;
611}

614template <class T>
615void Graph<T>::clear() {
616 // Delete the matrices first while we still know the original vertexSet.size()
617 deleteMatrix(distMatrix, vertexSet.size());
618 deleteMatrix(pathMatrix, vertexSet.size());
619 distMatrix = nullptr;
620 pathMatrix = nullptr;
621
622 // Delete all edges before deleting vertices. This matters because
623 // deleteEdge updates the destination vertex incoming list.
624 for (auto v : vertexSet) {
625 v->removeOutgoingEdges();
626 }
627 for (auto v : vertexSet) {
628 delete v;
629 }
630
631 // Empty the vector
632 vertexSet.clear();
633}

637bool Graph<T>::addEdge(const T &sourc, const T &dest, double w) {
638 auto v1 = findVertex(sourc);
639 auto v2 = findVertex(dest);
640 if (v1 == nullptr || v2 == nullptr)
641 return false;
642 for (auto edge : v1->getAdj()) {

660template <class T>
661bool Graph<T>::addBidirectionalEdge(const T &sourc, const T &dest, double w) {
662 auto v1 = findVertex(sourc);
663 auto v2 = findVertex(dest);
664 if (v1 == nullptr || v2 == nullptr)
665 return false;
666 bool existsForward = false;
667 bool existsReverse = false;
668 for (auto edge : v1->getAdj()) {
669 if (edge->getDest()->getInfo() == dest) existsForward = true;
670 }
671 for (auto edge : v2->getAdj()) {
672 if (edge->getDest()->getInfo() == sourc) existsReverse = true;
673 }
674 if (existsForward && existsReverse) return false;
675 Edge<T>* e1 = nullptr;
676 Edge<T>* e2 = nullptr;
677 if (!existsForward) e1 = v1->addEdge(v2, w);
678 if (!existsReverse) e2 = v2->addEdge(v1, w);
679 if (e1 != nullptr && e2 != nullptr) {
680 e1->setReverse(e2);
681 e2->setReverse(e1);
682 }
683 return true;
684}
